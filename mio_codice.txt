import os
import configparser
import shutil
from config import *

def clean_gt_for_soccer():
    # Percorso ai tuoi dati di test (o train se stai facendo prove)
    base_paths = [DATASET_TR23_TEST, DATASET_TR23_TRAIN]
    
    # Classi che consideriamo "TARGET" da tracciare
    # In base al PDF, il task ÃƒÂ¨ "Player tracking". 
    # Solitamente include portieri. Gli arbitri a volte sono esclusi o inclusi.
    # Se il gameinfo distingue "referee", decidi tu se tenerlo. 
    # Qui sotto tengo players, goalkeepers e referee.
    valid_keywords = ['player', 'goalkeeper', 'referee'] 

    
    for base_path in base_paths:
        if not os.path.exists(base_path):
            print(f"Errore: {base_path} non trovato.")
            return
        
        for seq_name in os.listdir(base_path):
            seq_path = os.path.join(base_path, seq_name)
            gameinfo_path = os.path.join(seq_path, 'gameinfo.ini')
            gt_folder = os.path.join(seq_path, 'gt')
            gt_path = os.path.join(gt_folder, 'gt.txt')
            
            # Facciamo un backup del gt originale se non esiste
            gt_backup = os.path.join(gt_folder, 'gt_backup.txt')

            if os.path.exists(gameinfo_path) and os.path.exists(gt_path):
                if not os.path.exists(gt_backup):
                    shutil.copy(gt_path, gt_backup)
                
                # Leggiamo gli ID da rimuovere (es. la palla)
                config = configparser.ConfigParser()
                config.read(gameinfo_path)
                
                ids_to_keep = []
                if 'Sequence' in config:
                    for key, value in config['Sequence'].items():
                        if key.startswith('trackletid_'):
                            # Se la descrizione contiene player/goalkeeper/referee, teniamo l'ID
                            if any(k in value.lower() for k in valid_keywords):
                                try:
                                    # trackletID_20 -> prendiamo 20
                                    obj_id = int(key.split('_')[1])
                                    ids_to_keep.append(obj_id)
                                except:
                                    pass

                # Riscriviamo il GT pulito
                with open(gt_backup, 'r') as f_in, open(gt_path, 'w') as f_out:
                    for line in f_in:
                        parts = line.strip().split(',')
                        # Formato GT: frame, ID, x, y, w, h, conf, class, vis
                        if len(parts) < 6: continue
                        
                        obj_id = int(parts[1])
                        
                        if obj_id in ids_to_keep:
                            # TRUCCO TECNICO:
                            # Assicuriamoci che ci siano 10 colonne (standard MOT)
                            # E impostiamo la colonna 8 (classe) a 1 ("pedestrian" per TrackEval)
                            # Anche se sono calciatori, TrackEval vuole vedere "1" lÃƒÂ¬.
                            
                            # Recuperiamo i dati essenziali
                            frame = parts[0]
                            pid = parts[1]
                            x, y, w, h = parts[2:6]
                            
                            # Creiamo la riga perfetta per TrackEval:
                            # conf=1, class=1, visibility=1
                            new_line = f"{frame},{pid},{x},{y},{w},{h},1,1,1\n"
                            f_out.write(new_line)
                
                print(f"Pulito GT per sequenza: {seq_name}")

if __name__ == "__main__":
    clean_gt_for_soccer()

def convert_dataset_from_manifest():
    print("--- Inizio Conversione Dataset (Manifest Mode) ---")
    
    # 1. Verifica esistenza file manifest
    train_manifest = SPLIT_DIR / "train.txt"
    val_manifest = SPLIT_DIR / "val.txt"
    
    if not train_manifest.exists() or not val_manifest.exists():
        print(f"[ERRORE] File manifest non trovati in {SPLIT_DIR}")
        print("Esegui prima 'src/Dataset/create_splits.py' per generare le liste!")
        return

    # 2. Leggi le liste delle sequenze
    with open(train_manifest, 'r') as f:
        train_seqs = [line.strip() for line in f if line.strip()]
        
    with open(val_manifest, 'r') as f:
        val_seqs = [line.strip() for line in f if line.strip()]
        
    print(f"Sequenze caricate -> Train: {len(train_seqs)}, Val: {len(val_seqs)}")

    # 3. Pulisci e Prepara le cartelle di destinazione (train/val)
    # Nota: Non cancelliamo SPLIT_DIR perchÃƒÂ© ÃƒÂ¨ dentro YOLO_DATASET_PATH
    images_dir = YOLO_DATASET_PATH / "images"
    labels_dir = YOLO_DATASET_PATH / "labels"
    
    if images_dir.exists(): shutil.rmtree(images_dir)
    if labels_dir.exists(): shutil.rmtree(labels_dir)
    
    for split in ['train', 'val']:
        (images_dir / split).mkdir(parents=True, exist_ok=True)
        (labels_dir / split).mkdir(parents=True, exist_ok=True)

    # ------------------------------------------------------------------
    # Funzione interna per processare una lista di sequenze
    # ------------------------------------------------------------------
    def process_split(seq_list, split_name):
        dest_img_dir = images_dir / split_name
        dest_lbl_dir = labels_dir / split_name
        
        print(f"Elaborazione split: {split_name}...")
        
        for seq_name in tqdm(seq_list):
            seq_path = DATASET_TR23_TRAIN / seq_name
            gt_path = seq_path / "gt" / "gt.txt"
            img_source_dir = seq_path / "img1"
            
            # Controllo integritÃƒÂ 
            if not seq_path.exists():
                print(f"[WARN] La sequenza '{seq_name}' ÃƒÂ¨ nel manifest ma non sul disco. Saltata.")
                continue
            if not gt_path.exists():
                # Alcune sequenze potrebbero non avere GT (raro nel train set), saltiamo
                continue

            # A. Otteniamo dimensioni immagine (necessario per normalizzare YOLO)
            # Leggiamo la prima immagine disponibile
            first_img = next(img_source_dir.glob("*.jpg"), None)
            if first_img is None:
                continue
                
            img_cv = cv2.imread(str(first_img))
            if img_cv is None: continue
            img_h, img_w, _ = img_cv.shape

            # B. Leggiamo e Parsiamo il GT (Ground Truth)
            # Formato MOT: frame, id, x_top, y_top, w, h, ...
            frames_data = {}
            with open(gt_path, 'r') as f:
                for line in f:
                    parts = line.strip().split(',')
                    if len(parts) < 6: continue
                    
                    frame_id = int(parts[0])
                    x_top, y_top, w_box, h_box = float(parts[2]), float(parts[3]), float(parts[4]), float(parts[5])
                    
                    # C. Convertiamo in formato YOLO (Normalizzato 0-1 e Centro)
                    # Class ID 0 = player
                    x_center = (x_top + w_box / 2) / img_w
                    y_center = (y_top + h_box / 2) / img_h
                    w_norm = w_box / img_w
                    h_norm = h_box / img_h
                    
                    # Clipping per sicurezza (valori devono stare tra 0 e 1)
                    x_center = max(0, min(1, x_center))
                    y_center = max(0, min(1, y_center))
                    w_norm = max(0, min(1, w_norm))
                    h_norm = max(0, min(1, h_norm))
                    
                    if frame_id not in frames_data:
                        frames_data[frame_id] = []
                    
                    # Stringa YOLO: class x_c y_c w h
                    frames_data[frame_id].append(f"0 {x_center:.6f} {y_center:.6f} {w_norm:.6f} {h_norm:.6f}")

            # D. Copiamo le immagini e creiamo i file label
            for img_file in img_source_dir.glob("*.jpg"):
                frame_idx = int(img_file.stem) # es. '000001' -> 1

                if split_name == 'train':
                    # Esempio: tiene frame 1, 11, 21... (se il resto della div ÃƒÂ¨ 1)
                    # Oppure frame 10, 20, 30... (se resto ÃƒÂ¨ 0). 
                    # Usiamo (frame_idx - 1) % RATE == 0 per prendere il primo frame sicuro.
                    if (frame_idx - 1) % TRAIN_SUBSAMPLE_RATE != 0:
                        continue 
                
                # Nome univoco: SequenceName_ImageName.jpg
                new_filename = f"{seq_name}_{img_file.name}"
                
                # Copia immagine
                shutil.copy(img_file, dest_img_dir / new_filename)
                
                # Crea file label (.txt)
                label_filename = f"{seq_name}_{img_file.stem}.txt"
                label_path = dest_lbl_dir / label_filename
                
                content = frames_data.get(frame_idx, [])
                
                with open(label_path, 'w') as lf:
                    if content:
                        lf.write("\n".join(content))
                    else:
                        # File vuoto = nessuna detection in questo frame (Background image)
                        # YOLO lo usa per capire dove NON c'ÃƒÂ¨ nulla
                        pass 

    # 4. Eseguiamo la conversione per entrambi gli split
    process_split(train_seqs, 'train')
    process_split(val_seqs, 'val')

    # 5. Creazione dataset.yaml per Ultralytics
    # Usiamo path assoluti o relativi al file yaml
    yaml_content = f"""
path: {YOLO_DATASET_PATH.as_posix()}
train: images/train
val: images/val
names:
  0: player
"""
    yaml_path = YOLO_DATASET_PATH / "dataset.yaml"
    with open(yaml_path, 'w') as f:
        f.write(yaml_content)

    print("------------------------------------------------")
    print("Conversione Completata!")
    print(f"Dataset YOLO pronto in: {YOLO_DATASET_PATH}")
    print(f"File configurazione: {yaml_path}")
    print("------------------------------------------------")

if __name__ == "__main__":
    convert_dataset_from_manifest()

import os

import torch

from PIL import Image

# Setup path
ROOT_DIR = Path(__file__).parent.parent.parent.resolve()

# Parametri
VAL_RATIO = 0.2
SEED = 42
FRAMES_PER_SEQ = 3  
MIN_CLUSTERS = 2
MAX_CLUSTERS = 10   

def extract_features(dataset_path):
    device = get_best_device()
    print(f"Caricamento modello ResNet18 su {device}...")
    
    model = models.resnet18(weights=models.ResNet18_Weights.IMAGENET1K_V1)
    model = nn.Sequential(*list(model.children())[:-1]) 
    model.to(device)
    model.eval()
    
    preprocess = transforms.Compose([
        transforms.Resize(256),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ])

    features_dict = {} # Mappa seq_name -> feature_vector
    
    seq_dirs = sorted([s for s in os.listdir(dataset_path) if (dataset_path / s).is_dir()])
    
    print(f"Estrazione feature da {len(seq_dirs)} sequenze...")
    
    with torch.no_grad():
        for seq in tqdm(seq_dirs):
            img_dir = dataset_path / seq / "img1"
            all_images = sorted(list(img_dir.glob("*.jpg")))
            
            if not all_images: continue
            
            indices = np.linspace(0, len(all_images) - 1, FRAMES_PER_SEQ, dtype=int)
            selected_imgs = [all_images[i] for i in indices]
            
            seq_feats = []
            for img_path in selected_imgs:
                try:
                    img = Image.open(img_path).convert('RGB')
                    img_t = preprocess(img).unsqueeze(0).to(device)
                    emb = model(img_t).flatten().cpu().numpy()
                    seq_feats.append(emb)
                except Exception:
                    pass

            if seq_feats:
                avg_feat = np.mean(seq_feats, axis=0)
                features_dict[seq] = avg_feat

    return features_dict

def find_best_k(features, min_k=2, max_k=10):
    print("\n--- Ricerca Automatica K (Clustering) ---")
    best_k = min_k
    best_score = -1
    best_labels = None
    
    limit_k = min(len(features), max_k)
    if len(features) < 2: return 1, np.zeros(len(features))

    for k in range(min_k, limit_k + 1):
        kmeans = KMeans(n_clusters=k, random_state=SEED, n_init=10)
        labels = kmeans.fit_predict(features)
        score = silhouette_score(features, labels)
        if score > best_score:
            best_score = score
            best_k = k
            best_labels = labels
            
    print(f"--> Miglior K trovato: {best_k} (Silhouette Score: {best_score:.4f})")
    return best_k, best_labels

def visualize_split_quality(features_dict, train_seqs, val_seqs):
    """
    Genera un grafico 2D per mostrare quanto Train e Val si sovrappongono.
    """
    print("\nGenerazione grafico di qualitÃƒÂ  dello split...")
    
    all_seqs = train_seqs + val_seqs
    X = np.array([features_dict[s] for s in all_seqs])
    
    # Riduciamo a 2 dimensioni con PCA per poter disegnare
    pca = PCA(n_components=2)
    X_pca = pca.fit_transform(X)
    
    # Creiamo due liste di coordinate
    train_indices = [i for i, s in enumerate(all_seqs) if s in train_seqs]
    val_indices = [i for i, s in enumerate(all_seqs) if s in val_seqs]
    
    plt.figure(figsize=(10, 8))
    
    # Disegna Train (Blu)
    plt.scatter(X_pca[train_indices, 0], X_pca[train_indices, 1], 
                c='blue', alpha=0.5, label='Train Set', s=50)
    
    # Disegna Val (Rosso)
    plt.scatter(X_pca[val_indices, 0], X_pca[val_indices, 1], 
                c='red', marker='x', alpha=0.9, label='Val Set', s=70)
    
    plt.title(f"Distribuzione Train vs Val (PCA)\nObiettivo: I punti rossi devono essere ben distribuiti tra i blu")
    plt.xlabel("Principal Component 1")
    plt.ylabel("Principal Component 2")
    plt.legend()
    plt.grid(True, alpha=0.3)
    
    output_plot = OUTPUT_DIR / "split_quality_check.png"
    if not OUTPUT_DIR.exists(): OUTPUT_DIR.mkdir(parents=True)
    
    plt.savefig(output_plot)
    print(f"--> Grafico salvato in: {output_plot}")
    print("    APRILO! Se i punti rossi coprono bene l'area dei blu, lo split ÃƒÂ¨ ottimo.")

def create_smart_splits():
    if not DATASET_TR23_TRAIN.exists():
        print(f"Dataset non trovato in {DATASET_TR23_TRAIN}")
        return

    # 1. Estrazione Features
    features_map = extract_features(DATASET_TR23_TRAIN)
    sequences = sorted(list(features_map.keys()))
    features_matrix = np.array([features_map[s] for s in sequences])
    
    if len(sequences) == 0: return

    # 2. Clustering
    n_clusters, clusters = find_best_k(features_matrix, MIN_CLUSTERS, MAX_CLUSTERS)
    
    clustered_seqs = defaultdict(list)
    for seq, cluster_id in zip(sequences, clusters):
        clustered_seqs[cluster_id].append(seq)
        
    # 3. Stratified Split
    train_final = []
    val_final = []
    
    print("\n--- Generazione Split ---")
    for cid in sorted(clustered_seqs.keys()):
        seqs_in_cluster = clustered_seqs[cid]
        random.seed(SEED)
        random.shuffle(seqs_in_cluster)
        
        n_val = int(len(seqs_in_cluster) * VAL_RATIO)
        if n_val == 0 and len(seqs_in_cluster) > 1: n_val = 1
            
        val_subset = seqs_in_cluster[:n_val]
        train_subset = seqs_in_cluster[n_val:]
        
        val_final.extend(val_subset)
        train_final.extend(train_subset)
        print(f"Cluster {cid}: {len(train_subset)} Train, {len(val_subset)} Val")

    # 4. Salva i file
    SPLIT_DIR.mkdir(parents=True, exist_ok=True)
    with open(SPLIT_DIR / "train.txt", "w") as f: f.write("\n".join(sorted(train_final)))
    with open(SPLIT_DIR / "val.txt", "w") as f: f.write("\n".join(sorted(val_final)))
    
    # 5. VERIFICA VISIVA (Nuovo!)
    visualize_split_quality(features_map, train_final, val_final)

if __name__ == "__main__":
    create_smart_splits()

def generate_manifests():

                        outputs,
                        inputs.input_ids,
                        target_sizes=[(height, width)]
                    )[0]

                    for score, label, box in zip(results["scores"], results["labels"], results["boxes"]):
                        
                        if score.item() < BOX_THRESHOLD:
                            continue
                        
                        if "person" in label:
                            x1, y1, x2, y2 = box.cpu().numpy()
                            w = x2 - x1
                            h = y2 - y1
                            conf = score.item()

                            # Scrittura formato MOT
                            # frame_id, id_obj(-1), x1, y1, w, h, conf, -1, -1, -1
                            line = f"{frame_id},-1,{x1:.2f},{y1:.2f},{w:.2f},{h:.2f},{conf:.4f},-1,1,-1\n"
                            f.write(line)
                            lines_written += 1
                            
                except Exception as e:
                    print(f"\nErrore nel frame {img_name}: {e}")
                    continue

            print(f"     COMPLETATO! Scritte {lines_written} righe.        ")
            files_created += 1

    print(f"\n--- FINE ---")
    print(f"File salvati in: {output_folder}")

if __name__ == "__main__":
    run_grounding_dino_detection()

    """

def run_detection_and_save_txt():
    # Carica il modello
    DEVICE = get_best_device()
    model_name = 'best.pt'  # Puoi cambiare con il percorso del tuo modello
    model = YOLO(MODEL_PATH / "training_runs" / "soccernet_high_precision" / "weights" / model_name) 
    model.to(DEVICE)
    print(f"Modello caricato su: {model.device}")

    base_path = DATASET_TR23_TEST
    output_folder = DETECTOR_OUTPUT_DIR / model_name.split(".")[0] # Cartella dove salveremo i txt per TrackEval
    
    if not os.path.exists(output_folder):
        os.makedirs(output_folder)
    else:
        # Se la cartella esiste giÃƒÂ , controlliamo se ÃƒÂ¨ vuota
        if len(os.listdir(output_folder)) > 0:
            print(f"\nATTENZIONE: La cartella di output '{output_folder}' contiene giÃƒÂ  dei file.")
            conferma = input("Vuoi eliminare TUTTI i file presenti e continuare? (y/n): ").strip().lower()

            if conferma in ['s', 'si', 'y', 'yes']:
                print("Eliminazione dei file in corso...")
                for filename in os.listdir(output_folder):
                    file_path = os.path.join(output_folder, filename)
                    try:
                        if os.path.isfile(file_path) or os.path.islink(file_path):
                            os.unlink(file_path) # Elimina file o link simbolici
                        elif os.path.isdir(file_path):
                            shutil.rmtree(file_path) # Elimina sottocartelle
                    except Exception as e:
                        print(f"Impossibile eliminare {file_path}. Motivo: {e}")
                print("Cartella svuotata con successo.\n")
            else:
                print("Operazione annullata. Esecuzione interrotta.")
                sys.exit() # Esce dallo script


    if not os.path.exists(base_path):
        print(f"Errore: Percorso {base_path} non trovato.")
        return

    sequences = os.listdir(base_path)

    for seq_name in sequences:
        seq_path = os.path.join(base_path, seq_name)
        img_source = os.path.join(seq_path, 'img1')

        if os.path.isdir(seq_path) and os.path.exists(img_source):
            print(f"--> Elaborazione sequenza: {seq_name}")
            
            # File di output per questa sequenza
            res_file_path = os.path.join(output_folder, f"{seq_name}.txt")
            
            # Eseguiamo l'inferenza usando lo stream per non saturare la RAM
            # stream=True restituisce un generatore
            results = model.predict(source=img_source, conf=0.25, imgsz=1280, verbose=False, stream=True, device=0, classes=[0])
            
            with open(res_file_path, 'w') as f:
                for result in results:
                    # Otteniamo il numero del frame dal percorso del file
                    # Es: '.../000001.jpg' -> frame_id = 1
                    path = result.path
                    frame_id = int(os.path.basename(path).split('.')[0])
                    
                    # I box sono in formato xywh (x_center, y_center, width, height) o xyxy
                    # Per MOT serve: Top-Left X, Top-Left Y, Width, Height
                    boxes = result.boxes.cpu().numpy()
                    
                    for box in boxes:
                        # Estrai coordinate xywh (top-left)
                        # box.xywh restituisce centro; box.xyxy restituisce angoli.
                        # box.xywhn normalizzato. 
                        # Ultralytics ha xywh come (center_x, center_y, w, h).
                        # Noi vogliamo (top_left_x, top_left_y, w, h).
                        
                        x1, y1, x2, y2 = box.xyxy[0]
                        w = x2 - x1
                        h = y2 - y1
                        conf = box.conf[0]
                        cls = box.cls[0]

                        # FILTRO: Assicuriamoci di salvare solo la classe "persona" (id 0 in COCO)
                        if int(cls) == 0: 
                            # Scriviamo nel formato MOT:
                            # frame, id, x, y, w, h, conf, -1, -1, -1
                            # Dato che ÃƒÂ¨ detection pura, ID mettiamo -1 o 1, non importa per DetA
                            line = f"{frame_id},-1,{x1:.2f},{y1:.2f},{w:.2f},{h:.2f},{conf:.4f},-1,1,-1\n"
                            f.write(line)
            
            print(f"Salvati risultati in: {res_file_path}")

            with open(SEQMAP_FILE, 'a') as seqmap_file:
                seqmap_file.write(f"{seq_name}\n")

if __name__ == "__main__":
    run_detection_and_save_txt()

def fix_ids_for_detection_only(trackers_folder):
    """
    Scansiona i file .txt dei risultati e assegna un ID univoco a ogni riga.
    NOTA: Questo distrugge le informazioni di tracking temporale (AssA sarÃƒÂ  ~0),
    ma ÃƒÂ¨ perfetto se ti interessa calcolare SOLO la DetA (Detection Accuracy).
    """
    import glob
    
    txt_files = glob.glob(os.path.join(trackers_folder, '**', '*.txt'), recursive=True)
    
    if not txt_files:
        print(f"Nessun file .txt trovato in {trackers_folder} da correggere.")
        return

    print(f"Correzione ID per {len(txt_files)} file (Detection only mode)...")
    
    for txt_file in txt_files:
        with open(txt_file, 'r') as f:
            lines = f.readlines()
        
        new_lines = []
        for i, line in enumerate(lines):
            parts = line.strip().split(',')
            if len(parts) < 6: continue
            
            # Assegna un ID progressivo unico per riga (rompe il tracking, ok per detection)
            parts[1] = str(i + 1) 
            
            new_lines.append(','.join(parts) + '\n')
            
        with open(txt_file, 'w') as f:
            f.writelines(new_lines)
    
    print("Correzione ID completata.")

import os
import configparser
import shutil
from config import *

def clean_gt_for_soccer():
    # Percorso ai tuoi dati di test (o train se stai facendo prove)
    base_paths = [DATASET_TR23_TEST, DATASET_TR23_TRAIN]
    
    # Classi che consideriamo "TARGET" da tracciare
    # In base al PDF, il task Ã¨ "Player tracking". 
    # Solitamente include portieri. Gli arbitri a volte sono esclusi o inclusi.
    # Se il gameinfo distingue "referee", decidi tu se tenerlo. 
    # Qui sotto tengo players, goalkeepers e referee.
    valid_keywords = ['player', 'goalkeeper', 'referee'] 

    
    for base_path in base_paths:
        if not os.path.exists(base_path):
            print(f"Errore: {base_path} non trovato.")
            return
        
        for seq_name in os.listdir(base_path):
            seq_path = os.path.join(base_path, seq_name)
            gameinfo_path = os.path.join(seq_path, 'gameinfo.ini')
            gt_folder = os.path.join(seq_path, 'gt')
            gt_path = os.path.join(gt_folder, 'gt.txt')
            
            # Facciamo un backup del gt originale se non esiste
            gt_backup = os.path.join(gt_folder, 'gt_backup.txt')

            if os.path.exists(gameinfo_path) and os.path.exists(gt_path):
                if not os.path.exists(gt_backup):
                    shutil.copy(gt_path, gt_backup)
                
                # Leggiamo gli ID da rimuovere (es. la palla)
                config = configparser.ConfigParser()
                config.read(gameinfo_path)
                
                ids_to_keep = []
                if 'Sequence' in config:
                    for key, value in config['Sequence'].items():
                        if key.startswith('trackletid_'):
                            # Se la descrizione contiene player/goalkeeper/referee, teniamo l'ID
                            if any(k in value.lower() for k in valid_keywords):
                                try:
                                    # trackletID_20 -> prendiamo 20
                                    obj_id = int(key.split('_')[1])
                                    ids_to_keep.append(obj_id)
                                except:
                                    pass

                # Riscriviamo il GT pulito
                with open(gt_backup, 'r') as f_in, open(gt_path, 'w') as f_out:
                    for line in f_in:
                        parts = line.strip().split(',')
                        # Formato GT: frame, ID, x, y, w, h, conf, class, vis
                        if len(parts) < 6: continue
                        
                        obj_id = int(parts[1])
                        
                        if obj_id in ids_to_keep:
                            # TRUCCO TECNICO:
                            # Assicuriamoci che ci siano 10 colonne (standard MOT)
                            # E impostiamo la colonna 8 (classe) a 1 ("pedestrian" per TrackEval)
                            # Anche se sono calciatori, TrackEval vuole vedere "1" lÃ¬.
                            
                            # Recuperiamo i dati essenziali
                            frame = parts[0]
                            pid = parts[1]
                            x, y, w, h = parts[2:6]
                            
                            # Creiamo la riga perfetta per TrackEval:
                            # conf=1, class=1, visibility=1
                            new_line = f"{frame},{pid},{x},{y},{w},{h},1,1,1\n"
                            f_out.write(new_line)
                
                print(f"Pulito GT per sequenza: {seq_name}")

if __name__ == "__main__":
    clean_gt_for_soccer()

import os
import shutil
import sys
import cv2
from pathlib import Path
from tqdm import tqdm

# Aggiungiamo la root al path per importare config
ROOT_DIR = Path(__file__).parent.parent.parent.resolve()
sys.path.append(str(ROOT_DIR))


# Definiamo il tasso di campionamento (1 frame ogni 25)
TRAIN_SUBSAMPLE_RATE = 25

# Importiamo le costanti definite in config
from config import DATASET_TR23_TRAIN, YOLO_DATASET_PATH, SPLIT_DIR

def convert_dataset_from_manifest():
    print("--- Inizio Conversione Dataset (Manifest Mode) ---")
    
    # 1. Verifica esistenza file manifest
    train_manifest = SPLIT_DIR / "train.txt"
    val_manifest = SPLIT_DIR / "val.txt"
    
    if not train_manifest.exists() or not val_manifest.exists():
        print(f"[ERRORE] File manifest non trovati in {SPLIT_DIR}")
        print("Esegui prima 'src/Dataset/create_splits.py' per generare le liste!")
        return

    # 2. Leggi le liste delle sequenze
    with open(train_manifest, 'r') as f:
        train_seqs = [line.strip() for line in f if line.strip()]
        
    with open(val_manifest, 'r') as f:
        val_seqs = [line.strip() for line in f if line.strip()]
        
    print(f"Sequenze caricate -> Train: {len(train_seqs)}, Val: {len(val_seqs)}")

    # 3. Pulisci e Prepara le cartelle di destinazione (train/val)
    # Nota: Non cancelliamo SPLIT_DIR perchÃ© Ã¨ dentro YOLO_DATASET_PATH
    images_dir = YOLO_DATASET_PATH / "images"
    labels_dir = YOLO_DATASET_PATH / "labels"
    
    if images_dir.exists(): shutil.rmtree(images_dir)
    if labels_dir.exists(): shutil.rmtree(labels_dir)
    
    for split in ['train', 'val']:
        (images_dir / split).mkdir(parents=True, exist_ok=True)
        (labels_dir / split).mkdir(parents=True, exist_ok=True)

    # ------------------------------------------------------------------
    # Funzione interna per processare una lista di sequenze
    # ------------------------------------------------------------------
    def process_split(seq_list, split_name):
        dest_img_dir = images_dir / split_name
        dest_lbl_dir = labels_dir / split_name
        
        print(f"Elaborazione split: {split_name}...")
        
        for seq_name in tqdm(seq_list):
            seq_path = DATASET_TR23_TRAIN / seq_name
            gt_path = seq_path / "gt" / "gt.txt"
            img_source_dir = seq_path / "img1"
            
            # Controllo integritÃ 
            if not seq_path.exists():
                print(f"[WARN] La sequenza '{seq_name}' Ã¨ nel manifest ma non sul disco. Saltata.")
                continue
            if not gt_path.exists():
                # Alcune sequenze potrebbero non avere GT (raro nel train set), saltiamo
                continue

            # A. Otteniamo dimensioni immagine (necessario per normalizzare YOLO)
            # Leggiamo la prima immagine disponibile
            first_img = next(img_source_dir.glob("*.jpg"), None)
            if first_img is None:
                continue
                
            img_cv = cv2.imread(str(first_img))
            if img_cv is None: continue
            img_h, img_w, _ = img_cv.shape

            # B. Leggiamo e Parsiamo il GT (Ground Truth)
            # Formato MOT: frame, id, x_top, y_top, w, h, ...
            frames_data = {}
            with open(gt_path, 'r') as f:
                for line in f:
                    parts = line.strip().split(',')
                    if len(parts) < 6: continue
                    
                    frame_id = int(parts[0])
                    x_top, y_top, w_box, h_box = float(parts[2]), float(parts[3]), float(parts[4]), float(parts[5])
                    
                    # C. Convertiamo in formato YOLO (Normalizzato 0-1 e Centro)
                    # Class ID 0 = player
                    x_center = (x_top + w_box / 2) / img_w
                    y_center = (y_top + h_box / 2) / img_h
                    w_norm = w_box / img_w
                    h_norm = h_box / img_h
                    
                    # Clipping per sicurezza (valori devono stare tra 0 e 1)
                    x_center = max(0, min(1, x_center))
                    y_center = max(0, min(1, y_center))
                    w_norm = max(0, min(1, w_norm))
                    h_norm = max(0, min(1, h_norm))
                    
                    if frame_id not in frames_data:
                        frames_data[frame_id] = []
                    
                    # Stringa YOLO: class x_c y_c w h
                    frames_data[frame_id].append(f"0 {x_center:.6f} {y_center:.6f} {w_norm:.6f} {h_norm:.6f}")

            # D. Copiamo le immagini e creiamo i file label
            for img_file in img_source_dir.glob("*.jpg"):
                frame_idx = int(img_file.stem) # es. '000001' -> 1

                if split_name == 'train':
                    # Esempio: tiene frame 1, 11, 21... (se il resto della div Ã¨ 1)
                    # Oppure frame 10, 20, 30... (se resto Ã¨ 0). 
                    # Usiamo (frame_idx - 1) % RATE == 0 per prendere il primo frame sicuro.
                    if (frame_idx - 1) % TRAIN_SUBSAMPLE_RATE != 0:
                        continue 
                
                # Nome univoco: SequenceName_ImageName.jpg
                new_filename = f"{seq_name}_{img_file.name}"
                
                # Copia immagine
                shutil.copy(img_file, dest_img_dir / new_filename)
                
                # Crea file label (.txt)
                label_filename = f"{seq_name}_{img_file.stem}.txt"
                label_path = dest_lbl_dir / label_filename
                
                content = frames_data.get(frame_idx, [])
                
                with open(label_path, 'w') as lf:
                    if content:
                        lf.write("\n".join(content))
                    else:
                        # File vuoto = nessuna detection in questo frame (Background image)
                        # YOLO lo usa per capire dove NON c'Ã¨ nulla
                        pass 

    # 4. Eseguiamo la conversione per entrambi gli split
    process_split(train_seqs, 'train')
    process_split(val_seqs, 'val')

    # 5. Creazione dataset.yaml per Ultralytics
    # Usiamo path assoluti o relativi al file yaml
    yaml_content = f"""
path: {YOLO_DATASET_PATH.as_posix()}
train: images/train
val: images/val
names:
  0: player

import os
import random
import sys
import numpy as np
import torch
import torch.nn as nn
from torchvision import models, transforms
from PIL import Image
from pathlib import Path
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score
from sklearn.decomposition import PCA
import matplotlib.pyplot as plt
from collections import defaultdict
from tqdm import tqdm

# Setup path
ROOT_DIR = Path(__file__).parent.parent.parent.resolve()
sys.path.append(str(ROOT_DIR))
from config import DATASET_TR23_TRAIN, SPLIT_DIR, OUTPUT_DIR
from src.utils.get_best_device import get_best_device

# Parametri
VAL_RATIO = 0.2
SEED = 42
FRAMES_PER_SEQ = 3  
MIN_CLUSTERS = 2
MAX_CLUSTERS = 10   

def extract_features(dataset_path):
    device = get_best_device()
    print(f"Caricamento modello ResNet18 su {device}...")
    
    model = models.resnet18(weights=models.ResNet18_Weights.IMAGENET1K_V1)
    model = nn.Sequential(*list(model.children())[:-1]) 
    model.to(device)
    model.eval()
    
    preprocess = transforms.Compose([
        transforms.Resize(256),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ])

    features_dict = {} # Mappa seq_name -> feature_vector
    
    seq_dirs = sorted([s for s in os.listdir(dataset_path) if (dataset_path / s).is_dir()])
    
    print(f"Estrazione feature da {len(seq_dirs)} sequenze...")
    
    with torch.no_grad():
        for seq in tqdm(seq_dirs):
            img_dir = dataset_path / seq / "img1"
            all_images = sorted(list(img_dir.glob("*.jpg")))
            
            if not all_images: continue
            
            indices = np.linspace(0, len(all_images) - 1, FRAMES_PER_SEQ, dtype=int)
            selected_imgs = [all_images[i] for i in indices]
            
            seq_feats = []
            for img_path in selected_imgs:
                try:
                    img = Image.open(img_path).convert('RGB')
                    img_t = preprocess(img).unsqueeze(0).to(device)
                    emb = model(img_t).flatten().cpu().numpy()
                    seq_feats.append(emb)
                except Exception:
                    pass

            if seq_feats:
                avg_feat = np.mean(seq_feats, axis=0)
                features_dict[seq] = avg_feat

    return features_dict

def find_best_k(features, min_k=2, max_k=10):
    print("\n--- Ricerca Automatica K (Clustering) ---")
    best_k = min_k
    best_score = -1
    best_labels = None
    
    limit_k = min(len(features), max_k)
    if len(features) < 2: return 1, np.zeros(len(features))

    for k in range(min_k, limit_k + 1):
        kmeans = KMeans(n_clusters=k, random_state=SEED, n_init=10)
        labels = kmeans.fit_predict(features)
        score = silhouette_score(features, labels)
        if score > best_score:
            best_score = score
            best_k = k
            best_labels = labels
            
    print(f"--> Miglior K trovato: {best_k} (Silhouette Score: {best_score:.4f})")
    return best_k, best_labels

def visualize_split_quality(features_dict, train_seqs, val_seqs):
    """
    Genera un grafico 2D per mostrare quanto Train e Val si sovrappongono.
    """
    print("\nGenerazione grafico di qualitÃ  dello split...")
    
    all_seqs = train_seqs + val_seqs
    X = np.array([features_dict[s] for s in all_seqs])
    
    # Riduciamo a 2 dimensioni con PCA per poter disegnare
    pca = PCA(n_components=2)
    X_pca = pca.fit_transform(X)
    
    # Creiamo due liste di coordinate
    train_indices = [i for i, s in enumerate(all_seqs) if s in train_seqs]
    val_indices = [i for i, s in enumerate(all_seqs) if s in val_seqs]
    
    plt.figure(figsize=(10, 8))
    
    # Disegna Train (Blu)
    plt.scatter(X_pca[train_indices, 0], X_pca[train_indices, 1], 
                c='blue', alpha=0.5, label='Train Set', s=50)
    
    # Disegna Val (Rosso)
    plt.scatter(X_pca[val_indices, 0], X_pca[val_indices, 1], 
                c='red', marker='x', alpha=0.9, label='Val Set', s=70)
    
    plt.title(f"Distribuzione Train vs Val (PCA)\nObiettivo: I punti rossi devono essere ben distribuiti tra i blu")
    plt.xlabel("Principal Component 1")
    plt.ylabel("Principal Component 2")
    plt.legend()
    plt.grid(True, alpha=0.3)
    
    output_plot = OUTPUT_DIR / "split_quality_check.png"
    if not OUTPUT_DIR.exists(): OUTPUT_DIR.mkdir(parents=True)
    
    plt.savefig(output_plot)
    print(f"--> Grafico salvato in: {output_plot}")
    print("    APRILO! Se i punti rossi coprono bene l'area dei blu, lo split Ã¨ ottimo.")

def create_smart_splits():
    if not DATASET_TR23_TRAIN.exists():
        print(f"Dataset non trovato in {DATASET_TR23_TRAIN}")
        return

    # 1. Estrazione Features
    features_map = extract_features(DATASET_TR23_TRAIN)
    sequences = sorted(list(features_map.keys()))
    features_matrix = np.array([features_map[s] for s in sequences])
    
    if len(sequences) == 0: return

    # 2. Clustering
    n_clusters, clusters = find_best_k(features_matrix, MIN_CLUSTERS, MAX_CLUSTERS)
    
    clustered_seqs = defaultdict(list)
    for seq, cluster_id in zip(sequences, clusters):
        clustered_seqs[cluster_id].append(seq)
        
    # 3. Stratified Split
    train_final = []
    val_final = []
    
    print("\n--- Generazione Split ---")
    for cid in sorted(clustered_seqs.keys()):
        seqs_in_cluster = clustered_seqs[cid]
        random.seed(SEED)
        random.shuffle(seqs_in_cluster)
        
        n_val = int(len(seqs_in_cluster) * VAL_RATIO)
        if n_val == 0 and len(seqs_in_cluster) > 1: n_val = 1
            
        val_subset = seqs_in_cluster[:n_val]
        train_subset = seqs_in_cluster[n_val:]
        
        val_final.extend(val_subset)
        train_final.extend(train_subset)
        print(f"Cluster {cid}: {len(train_subset)} Train, {len(val_subset)} Val")

    # 4. Salva i file
    SPLIT_DIR.mkdir(parents=True, exist_ok=True)
    with open(SPLIT_DIR / "train.txt", "w") as f: f.write("\n".join(sorted(train_final)))
    with open(SPLIT_DIR / "val.txt", "w") as f: f.write("\n".join(sorted(val_final)))
    
    # 5. VERIFICA VISIVA (Nuovo!)
    visualize_split_quality(features_map, train_final, val_final)

if __name__ == "__main__":
    create_smart_splits()

def generate_manifests():
    print(f"Generazione manifest partendo da: {DATASET_TR23_TRAIN}")
    
    if not DATASET_TR23_TRAIN.exists():
        print(f"ERRORE: La cartella dataset {DATASET_TR23_TRAIN} non esiste.")
        return

    # 1. Leggi tutte le cartelle (sequenze)
    sequences = [s for s in os.listdir(DATASET_TR23_TRAIN) if (DATASET_TR23_TRAIN / s).is_dir()]

    # 3. Mescola in modo deterministico
    random.seed(SEED)
    random.shuffle(sequences)
    
    # 4. Calcola il punto di taglio

    train_seqs = sequences[:split_idx]
    val_seqs = sequences[split_idx:]

    val_file = SPLIT_DIR / "val.txt"
    
    with open(train_file, "w") as f:
        f.write("\n".join(train_seqs))
        
    with open(val_file, "w") as f:
        f.write("\n".join(val_seqs))
        
    print("------------------------------------------------")
    print(f"Manifest creati con successo in: {SPLIT_DIR}")
    print(f"Train sequences ({len(train_seqs)}): salvate in train.txt")

        image_files = sorted([f for f in os.listdir(img_source) if f.lower().endswith(('.jpg', '.jpeg', '.png'))])
        num_imgs = len(image_files)
        
        if num_imgs == 0:
            print(f"  -> SALTATA {seq_name}: Cartella vuota")
            continue

        print(f"\nAnalisi sequenza: {seq_name} ({num_imgs} immagini)")
        
        res_file_path = os.path.join(output_folder, f"{seq_name}.txt")
        
        with open(res_file_path, 'w') as f:

                    continue

                    with torch.no_grad():
                        outputs = model(**inputs)

                    width, height = image_pil.size
                    
                    results = processor.post_process_grounded_object_detection(
                        outputs,
                        inputs.input_ids,

                            # Scrittura formato MOT

                            line = f"{frame_id},-1,{x1:.2f},{y1:.2f},{w:.2f},{h:.2f},{conf:.4f},-1,1,-1\n"
                            f.write(line)
                            lines_written += 1

            files_created += 1

    print(f"\n--- FINE ---")
    print(f"File salvati in: {output_folder}")

def fix_ids_for_detection_only(trackers_folder):
    """
    Scansiona i file .txt dei risultati e assegna un ID univoco a ogni riga.
    NOTA: Questo distrugge le informazioni di tracking temporale (AssA sarÃ  ~0),
    ma Ã¨ perfetto se ti interessa calcolare SOLO la DetA (Detection Accuracy).
    """
    import glob
    
    txt_files = glob.glob(os.path.join(trackers_folder, '**', '*.txt'), recursive=True)
    
    if not txt_files:
        print(f"Nessun file .txt trovato in {trackers_folder} da correggere.")
        return

    print(f"Correzione ID per {len(txt_files)} file (Detection only mode)...")
    
    for txt_file in txt_files:
        with open(txt_file, 'r') as f:
            lines = f.readlines()
        
        new_lines = []
        for i, line in enumerate(lines):
            parts = line.strip().split(',')
            if len(parts) < 6: continue
            
            # Assegna un ID progressivo unico per riga (rompe il tracking, ok per detection)
            parts[1] = str(i + 1) 
            
            new_lines.append(','.join(parts) + '\n')
            
        with open(txt_file, 'w') as f:
            f.writelines(new_lines)
    
    print("Correzione ID completata.")
